cmake_minimum_required(VERSION 3.16)
project(yzbot_tf2_runtime_fix)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(tf2 REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(console_bridge REQUIRED)
add_library(tf2_fixed SHARED buffer_core.cpp cache.cpp static_cache.cpp time.cpp)
target_compile_definitions(tf2_fixed PRIVATE TF2_BUILDING_DLL)
target_include_directories(tf2_fixed PRIVATE ${tf2_INCLUDE_DIRS} ${console_bridge_INCLUDE_DIRS})
ament_target_dependencies(tf2_fixed geometry_msgs builtin_interfaces)
target_link_libraries(tf2_fixed ${console_bridge_LIBRARIES})
set_target_properties(tf2_fixed PROPERTIES OUTPUT_NAME tf2)
add_executable(lock_order_regression lock_order_regression.cpp)
target_include_directories(lock_order_regression PRIVATE ${tf2_INCLUDE_DIRS})
ament_target_dependencies(lock_order_regression geometry_msgs builtin_interfaces)
target_link_libraries(lock_order_regression tf2_fixed pthread)
